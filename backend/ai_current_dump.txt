"""Provider-neutral location abstraction for cameras and response units."""

from __future__ import annotations

import os
from datetime import datetime, timezone
from typing import Any


SOURCES = {"CAMERA", "GPS", "MANUAL", "SIMULATED", "UNAVAILABLE"}


def unavailable_location() -> dict[str, Any]:
    return {"latitude": None, "longitude": None, "accuracy": None, "source": "UNAVAILABLE", "timestamp": None}


class LocationProvider:
    name = "unavailable"

    def get_location(self, subject_id: str) -> dict[str, Any]:
        return unavailable_location()


class SimulatedLocationProvider(LocationProvider):
    name = "simulation"

    def get_location(self, subject_id: str) -> dict[str, Any]:
        # No coordinates are invented; simulation is explicitly unavailable unless configured.
        return {**unavailable_location(), "source": "SIMULATED", "timestamp": datetime.now(timezone.utc).isoformat()}


class ManualLocationProvider(LocationProvider):
    name = "manual"

    def get_location(self, subject_id: str) -> dict[str, Any]:
        return unavailable_location()


class GPSLocationProvider(LocationProvider):
    name = "gps"

    def get_location(self, subject_id: str) -> dict[str, Any]:
        return unavailable_location()


def get_location_provider() -> LocationProvider:
    mode = os.getenv("GPS_MODE", "simulation").strip().lower()
    if mode == "gps":
        return GPSLocationProvider()
    if mode == "manual":
        return ManualLocationProvider()
    if mode == "simulation":
        return SimulatedLocationProvider()
    return LocationProvider()


def location_status() -> dict[str, Any]:
    provider = get_location_provider()
    return {
        "mode": os.getenv("GPS_MODE", "simulation").strip().lower(),
        "provider": provider.name,
        "available": provider.name in {"gps", "manual", "simulation"},
        "description": "Provider configured; coordinates are unavailable" if provider.name != "unavailable" else "No location provider configured",
    }

"""Safe dispatch provider abstraction.

The default provider only updates local simulated unit state. It never contacts
real emergency services or claims that a real unit was dispatched.
"""

from __future__ import annotations

import os
from datetime import datetime, timezone
from typing import Any


class DispatchProvider:
    name = "unavailable"

    def dispatch(self, response_id: str, unit_id: str, eta_minutes: int | None = None) -> dict[str, Any]:
        raise NotImplementedError


class SimulationDispatchProvider(DispatchProvider):
    name = "simulation"

    def dispatch(self, response_id: str, unit_id: str, eta_minutes: int | None = None) -> dict[str, Any]:
        return {
            "response_id": response_id,
            "unit_id": unit_id,
            "status": "DISPATCHED",
            "eta_minutes": eta_minutes,
            "mode": "SIMULATED",
            "timestamp": datetime.now(timezone.utc).isoformat(),
        }


class ExternalDispatchProvider(DispatchProvider):
    name = "external"

    def dispatch(self, response_id: str, unit_id: str, eta_minutes: int | None = None) -> dict[str, Any]:
        raise RuntimeError("External dispatch provider is not configured")


def get_dispatch_provider() -> DispatchProvider:
    mode = os.getenv("DISPATCH_MODE", "simulation").strip().lower()
    if mode == "external":
        return ExternalDispatchProvider()
    if mode == "simulation":
        return SimulationDispatchProvider()
    return DispatchProvider()


def dispatch_status() -> dict[str, Any]:
    provider = get_dispatch_provider()
    return {
        "mode": os.getenv("DISPATCH_MODE", "simulation").strip().lower(),
        "provider": provider.name,
        "safe_simulation": provider.name == "simulation",
        "description": "Local simulated dispatch only" if provider.name == "simulation" else "External dispatch is not configured",
    }

def _init_security_db() -> None:
    """Create audit_events + security_alerts tables if not present."""
    with get_connection() as conn:
        cursor = conn.cursor()
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS audit_events (
                id TEXT PRIMARY KEY,
                timestamp TEXT NOT NULL,
                user_id TEXT NOT NULL DEFAULT 'anonymous',
                user_role TEXT NOT NULL DEFAULT 'VIEWER',
                event_type TEXT NOT NULL,
                resource_type TEXT,
                resource_id TEXT,
                action TEXT,
                status TEXT NOT NULL DEFAULT 'SUCCESS',
                severity TEXT NOT NULL DEFAULT 'INFO',
                source_ip TEXT,
                metadata TEXT,
                created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
            )
        """)
        cursor.execute("CREATE INDEX IF NOT EXISTS idx_audit_timestamp ON audit_events(timestamp DESC)")
        cursor.execute("CREATE INDEX IF NOT EXISTS idx_audit_event_type ON audit_events(event_type)")
        cursor.execute("CREATE INDEX IF NOT EXISTS idx_audit_user_id ON audit_events(user_id)")
        cursor.execute("CREATE INDEX IF NOT EXISTS idx_audit_resource_id ON audit_events(resource_id)")

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS security_alerts (
                id TEXT PRIMARY KEY,
                timestamp TEXT NOT NULL,
                severity TEXT NOT NULL,
                source TEXT NOT NULL,
                reason TEXT NOT NULL,
                status TEXT NOT NULL DEFAULT 'ACTIVE',
                details TEXT,
                acknowledged_at TEXT,
                acknowledged_by TEXT,
                resolved_at TEXT,
                resolved_by TEXT
            )
        """)
        cursor.execute("CREATE INDEX IF NOT EXISTS idx_alerts_timestamp ON security_alerts(timestamp DESC)")
        cursor.execute("CREATE INDEX IF NOT EXISTS idx_alerts_status ON security_alerts(status)")

        conn.commit()

_SENSITIVE_KEYS = {
    "password", "token", "secret", "api_key", "api-key", "apikey",
    "authorization", "auth", "jwt", "bearer", "cookie", "session",
    "x-api-key", "x-auth-token", "credential", "credentials",

}


def _sanitize_metadata(meta: dict[str, Any] | None) -> dict[str, Any] | None:
    """Strip sensitive keys and truncate long string values."""
    if meta is None:
        return None
    if not isinstance(meta, dict):
        return {"value": str(meta)[:200]}
    out: dict[str, Any] = {}
    for k, v in meta.items():
        if any(s in k.lower() for s in _SENSITIVE_KEYS):
            out[k] = "[REDACTED]"
        elif isinstance(v, str):
            out[k] = v[:300]
        else:
            out[k] = v
    return out


def _sanitize_query(query: str) -> str:
    """Sanitize AI query before storage (truncate, strip credentials)."""
    if not query:
        return ""
    text = re.sub(r"\s+", " ", query).strip()
    if len(text) > 240:
        text = text[:240] + "â€¦"
    return text


def _next_id(prefix: str) -> str:
    import uuid
    return f"{prefix}_{uuid.uuid4().hex[:12]}"


def log_audit(
    event_type: str,
    user_id: str = "anonymous",
    user_role: str = ROLE_VIEWER,
    resource_type: str | None = None,
    resource_id: str | None = None,
    action: str | None = None,
    status: str = "SUCCESS",
    severity: str = "INFO",
    source_ip: str | None = None,
    metadata: dict[str, Any] | None = None,
) -> str | None:
    """Record an audit event. Returns the event id."""
    if event_type not in ALL_AUDIT_EVENT_TYPES:
        event_type = "OTHER"
    try:
        eid = _next_id("aud")
        now = datetime.now(timezone.utc).isoformat()
        meta_json = json.dumps(_sanitize_metadata(metadata) or {})
        with get_connection() as conn:
            cursor = conn.cursor()
            cursor.execute(
                """
                INSERT INTO audit_events
                  (id, timestamp, user_id, user_role, event_type, resource_type, resource_id,
                   action, status, severity, source_ip, metadata)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
                """,
                (eid, now, user_id, user_role, event_type, resource_type, resource_id,
                 action, status, severity, source_ip, meta_json),
            )
            conn.commit()
        return eid
    except Exception as e:
        print(f"[SentinelAI] âš ï¸ Audit log error: {e}")
        return None

def query_audit(
    limit: int = 50,
    offset: int = 0,
    event_type: str | None = None,
    user_id: str | None = None,
    resource_id: str | None = None,
    severity: str | None = None,
    status: str | None = None,
    since: str | None = None,
    until: str | None = None,
    search: str | None = None,
) -> dict[str, Any]:
    """Filter the audit log. Always paginated (default 50)."""
    limit = max(1, min(500, int(limit or 50)))
    offset = max(0, int(offset or 0))

    where: list[str] = []
    params: list[Any] = []

    if event_type:
        where.append("event_type = ?")
        params.append(event_type)
    if user_id:
        where.append("user_id = ?")
        params.append(user_id)
    if resource_id:
        where.append("resource_id = ?")
        params.append(resource_id)
    if severity:
        where.append("severity = ?")
        params.append(severity)
    if status:
        where.append("status = ?")
        params.append(status)
    if since:
        where.append("timestamp >= ?")
        params.append(since)
    if until:
        where.append("timestamp <= ?")
        params.append(until)
    if search:
        like = f"%{search[:80]}%"
        where.append("(action LIKE ? OR metadata LIKE ? OR resource_id LIKE ? OR user_id LIKE ?)")
        params.extend([like, like, like, like])

    where_sql = ("WHERE " + " AND ".join(where)) if where else ""

    try:
        with get_connection() as conn:
            cursor = conn.cursor()
            cursor.execute(
                f"SELECT COUNT(*) AS n FROM audit_events {where_sql}", params
            )
            total = cursor.fetchone()["n"]

            cursor.execute(
                f"""SELECT id, timestamp, user_id, user_role, event_type,
                           resource_type, resource_id, action, status, severity,
                           source_ip, metadata
                    FROM audit_events {where_sql}
                    ORDER BY timestamp DESC LIMIT ? OFFSET ?""",
                (*params, limit, offset),
            )
            events = []
            for r in cursor.fetchall():
                events.append({
                    "id": r["id"],
                    "timestamp": r["timestamp"],
                    "user_id": r["user_id"],
                    "user_role": r["user_role"],
                    "event_type": r["event_type"],
                    "resource_type": r["resource_type"],
                    "resource_id": r["resource_id"],
                    "action": r["action"],
                    "status": r["status"],
                    "severity": r["severity"],
                    "source_ip": r["source_ip"],
                    "metadata": _safe_json_loads(r["metadata"]),
                })

            # Counts by event type
            cursor.execute(
                "SELECT event_type, COUNT(*) AS n FROM audit_events GROUP BY event_type"
            )
            by_type = {row["event_type"]: row["n"] for row in cursor.fetchall()}

            # Counts by user
            cursor.execute(
                "SELECT user_id, COUNT(*) AS n FROM audit_events GROUP BY user_id"
            )
            by_user = {row["user_id"]: row["n"] for row in cursor.fetchall()}

        return {
            "events": events,
            "count": len(events),
            "total": total,
            "limit": limit,
            "offset": offset,
            "stats": {
                "by_type": by_type,
                "by_user": by_user,
            },
        }
    except Exception as e:
        print(f"[SentinelAI] âš ï¸ Audit query error: {e}")
        return {"events": [], "count": 0, "total": 0, "error": str(e)}

def track_failed_request(source_ip: str | None) -> None:
    """Track a failed request for threshold-based alerting."""
    if not source_ip:
        return
    now = time.time()
    with _security_lock:
        bucket = _failed_request_counts[source_ip]
        bucket.append(now)
        # Trim to window
        bucket[:] = [t for t in bucket if now - t <= RATE_WINDOW_SECONDS]
        if len(bucket) >= FAILED_REQUEST_THRESHOLD:
            _maybe_create_alert_locked(
                severity="HIGH",
                source=source_ip,
                reason=(
                    f"{len(bucket)} failed API requests in the last "
                    f"{RATE_WINDOW_SECONDS // 60} minutes from {source_ip}."
                ),
                details={"count": len(bucket), "window_seconds": RATE_WINDOW_SECONDS},
                dedup_key=f"failed_requests:{source_ip}",
            )


def track_failed_login(source_ip: str | None, username: str | None = None) -> None:
    """Track a failed login attempt."""
    if not source_ip:
        return
    now = time.time()
    with _security_lock:
        bucket = _failed_login_counts[source_ip]
        bucket.append(now)
        bucket[:] = [t for t in bucket if now - t <= RATE_WINDOW_SECONDS]
        if len(bucket) >= FAILED_LOGIN_THRESHOLD:
            _maybe_create_alert_locked(
                severity="CRITICAL",
                source=source_ip,
                reason=(
                    f"{len(bucket)} failed authentication attempts in the last "
                    f"{RATE_WINDOW_SECONDS // 60} minutes from {source_ip}."
                ),
                details={
                    "count": len(bucket),
                    "window_seconds": RATE_WINDOW_SECONDS,
                    "username_hint": (username[:8] + "â€¦") if username else None,
                },
                dedup_key=f"failed_login:{source_ip}",
            )


def _maybe_create_alert_locked(severity: str, source: str, reason: str,
                                 details: dict | None, dedup_key: str) -> None:
    """Create a security alert if not recently duplicated. Caller holds the lock."""
    now = time.time()
    last = _recent_alert_keys.get(dedup_key, 0)
    if now - last < ALERT_DEDUP_WINDOW_SECONDS:
        return
    _recent_alert_keys[dedup_key] = now
    try:
        aid = _next_id("sa")
        ts = datetime.now(timezone.utc).isoformat()
        with get_connection() as conn:
            cursor = conn.cursor()
            cursor.execute(
                """INSERT INTO security_alerts
                     (id, timestamp, severity, source, reason, status, details)
                   VALUES (?, ?, ?, ?, ?, ?, ?)""",
                (aid, ts, severity, source, reason, "ACTIVE",
                 json.dumps(details or {})),
            )
            conn.commit()
        # Also log an audit event for the alert itself
        log_audit(
            event_type=AUDIT_SECURITY_ALERT,
            user_id="system",
            user_role=ROLE_ADMIN,
            action=f"Auto-detected: {reason[:80]}",
            resource_type="security_alert",
            resource_id=aid,
            status="AUTO",
            severity=severity,
            metadata={"source": source, **(details or {})},
        )
    except Exception as e:
        print(f"[SentinelAI] âš ï¸ Security alert create error: {e}")

def acknowledge_security_alert(alert_id: str, actor: str) -> dict[str, Any] | None:
    try:
        now = datetime.now(timezone.utc).isoformat()
        with get_connection() as conn:
            cursor = conn.cursor()
            cursor.execute(
                """UPDATE security_alerts
                   SET status = 'ACKNOWLEDGED', acknowledged_at = ?, acknowledged_by = ?
                   WHERE id = ? AND status = 'ACTIVE'""",
                (now, actor, alert_id),
            )
            conn.commit()
            if cursor.rowcount == 0:
                return None
        log_audit(
            event_type=AUDIT_SECURITY_ALERT_ACK,
            user_id=actor,
            user_role=ROLE_ADMIN,
            resource_type="security_alert",
            resource_id=alert_id,
            action="acknowledge",
        )
        return {"id": alert_id, "status": "ACKNOWLEDGED", "acknowledged_at": now, "acknowledged_by": actor}
    except Exception:
        return None


def resolve_security_alert(alert_id: str, actor: str) -> dict[str, Any] | None:
    try:
        now = datetime.now(timezone.utc).isoformat()
        with get_connection() as conn:
            cursor = conn.cursor()
            cursor.execute(
                """UPDATE security_alerts
                   SET status = 'RESOLVED', resolved_at = ?, resolved_by = ?
                   WHERE id = ? AND status != 'RESOLVED'""",
                (now, actor, alert_id),
            )
            conn.commit()
            if cursor.rowcount == 0:
                return None
        log_audit(
            event_type=AUDIT_SECURITY_ALERT_RESOLVE,
            user_id=actor,
            user_role=ROLE_ADMIN,
            resource_type="security_alert",
            resource_id=alert_id,
            action="resolve",
        )
        return {"id": alert_id, "status": "RESOLVED", "resolved_at": now, "resolved_by": actor}
    except Exception:
        return None

def get_configuration_status() -> dict[str, Any]:
    """Return SECURITY/WARNING/CRITICAL for each configuration dimension.

    NEVER returns secret values, only booleans/levels."""
    import os

    checks: list[dict[str, Any]] = []

    # 1. Authentication foundation
    auth_enabled = bool(os.getenv("AUTH_ENABLED", "true").lower() in ("1", "true", "yes"))
    auth_secret_configured = len(os.getenv("SENTINEL_AUTH_SECRET", "")) >= 32
    checks.append({
        "key": "authentication_enabled",
        "label": "Authentication enabled",
        "status": "SECURE" if auth_enabled else "WARNING",
        "description": "Bearer authentication and role enforcement are enabled." if auth_enabled else "Authentication is disabled by environment configuration.",

def compute_security_score() -> dict[str, Any]:
    """Compute a 0-100 security health score.

    Factors:
      - Active security alerts (CRITICAL=âˆ’20, HIGH=âˆ’10, MEDIUM=âˆ’5, LOW=âˆ’2)
      - Recent failed requests
      - Configuration status
      - Backend health (always ONLINE in this build)
      - Total resolved alerts (positive)
    """
    try:
        alerts = list_security_alerts(limit=200)["alerts"]
        active_alerts = [a for a in alerts if a.get("status") == "ACTIVE"]
        resolved = [a for a in alerts if a.get("status") == "RESOLVED"]

        score = 100
        factors: list[dict[str, Any]] = []

        # Alerts deduction
        severity_penalty = {"CRITICAL": 20, "HIGH": 10, "MEDIUM": 5, "LOW": 2}
        for a in active_alerts:
            sev = a.get("severity", "LOW")
            score -= severity_penalty.get(sev, 2)
            factors.append({
                "factor": f"active_alert_{sev.lower()}",
                "delta": -severity_penalty.get(sev, 2),
                "reason": a.get("reason", "")[:80],
            })

        # Configuration penalty
        config = get_configuration_status()
        for c in config["checks"]:
            if c["status"] == "CRITICAL":
                score -= 15
                factors.append({"factor": f"config_{c['key']}", "delta": -15, "reason": c["label"]})
            elif c["status"] == "WARNING":
                score -= 5
                factors.append({"factor": f"config_{c['key']}", "delta": -5, "reason": c["label"]})

        # Positive: resolved alerts (each +1, capped +5)
        resolved_bonus = min(5, len(resolved))
        score += resolved_bonus
        if resolved_bonus:
            factors.append({"factor": "resolved_alerts", "delta": resolved_bonus, "reason": f"{len(resolved)} alert(s) resolved"})

        score = max(0, min(100, score))

        if score >= 80:
            category = "SECURE"
        elif score >= 50:
            category = "WARNING"
        else:
            category = "CRITICAL"

        return {
            "score": score,
            "category": category,
            "factors": factors,
            "active_alerts": len(active_alerts),
            "resolved_alerts": len(resolved),
            "generated_at": datetime.now(timezone.utc).isoformat(),
        }
    except Exception as e:
        return {"score": 50, "category": "WARNING", "factors": [{"factor": "compute_error", "delta": 0, "reason": str(e)}]}

def apply_retention(retention_days: int | None = None) -> dict[str, Any]:
    """Delete audit events older than retention_days. NOT auto-called.

    Caller must opt in explicitly. Returns the count of deleted rows.
    """
    days = int(retention_days or DEFAULT_RETENTION_DAYS)
    if days <= 0:
        return {"deleted": 0, "retention_days": days, "applied": False}
    try:
        cutoff = (datetime.now(timezone.utc) - timedelta(days=days)).isoformat()
        with get_connection() as conn:
            cursor = conn.cursor()
            cursor.execute("DELETE FROM audit_events WHERE timestamp < ?", (cutoff,))
            deleted = cursor.rowcount
            conn.commit()
        return {"deleted": deleted, "retention_days": days, "applied": True, "cutoff": cutoff}
    except Exception as e:
        return {"deleted": 0, "retention_days": days, "applied": False, "error": str(e)}

def _load_local_env() -> None:
    """Load backend/.env without overriding explicitly exported variables."""
    env_path = Path(__file__).resolve().parent.parent / ".env"
    if not env_path.exists():
        return
    try:
        for raw_line in env_path.read_text(encoding="utf-8").splitlines():

            key = key.strip()
            value = value.strip().strip('"').strip("'")
            if key and key not in os.environ:
                os.environ[key] = value

def _hash_password(password: str) -> str:
    salt = secrets.token_bytes(16)
    digest = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt, PBKDF2_ITERATIONS)
    return f"pbkdf2_sha256${PBKDF2_ITERATIONS}${_b64(salt)}${_b64(digest)}"


def _verify_password(password: str, encoded: str) -> bool:
    try:
        algorithm, iterations, salt, expected = encoded.split("$", 3)
        if algorithm != "pbkdf2_sha256":
            return False
        digest = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), _unb64(salt), int(iterations))
        return hmac.compare_digest(_b64(digest), expected)
    except (ValueError, TypeError):
        return False

def issue_token(user: dict[str, Any]) -> str:
    secret = token_secret()
    if len(secret) < 32:
        raise RuntimeError("SENTINEL_AUTH_SECRET must be at least 32 characters")
    header = _b64(json.dumps({"alg": "HS256", "typ": "JWT"}, separators=(",", ":")).encode())
    payload = _b64(json.dumps({"sub": user["id"], "username": user["username"], "role": user["role"], "exp": int(time.time()) + TOKEN_TTL_SECONDS}, separators=(",", ":")).encode())
    message = f"{header}.{payload}".encode("ascii")
    signature = _b64(hmac.new(secret.encode(), message, hashlib.sha256).digest())
    return f"{header}.{payload}.{signature}"


def validate_token(token: str) -> dict[str, Any] | None:
    secret = token_secret()
    if len(secret) < 32:
        return None
    try:
        header, payload, signature = token.split(".", 2)
        message = f"{header}.{payload}".encode("ascii")
        expected = _b64(hmac.new(secret.encode(), message, hashlib.sha256).digest())
        if not hmac.compare_digest(signature, expected):
            return None
        claims = json.loads(_unb64(payload))
        if int(claims.get("exp", 0)) <= int(time.time()):
            return None
        user = get_user_by_username(str(claims.get("username", "")))
        if not user or not user["is_active"] or user["role"] != claims.get("role"):
            return None
        return user
    except (ValueError, TypeError, json.JSONDecodeError):
        return None

def _score_incident(incident: dict[str, Any]) -> dict[str, Any]:
    incident_type = str(incident.get("incident_type") or "").upper()
    severity = str(incident.get("severity") or "LOW").upper()
    risk_score = max(0, min(100, int(incident.get("risk_score") or 0)))
    objects = incident.get("detected_objects") or []
    object_names = " ".join(str(item.get("class", "")) if isinstance(item, dict) else str(item) for item in objects).lower()

    score = risk_score * 0.62
    factors: list[dict[str, Any]] = [{"label": "Risk engine score", "weight": round(risk_score * 0.62)}]
    if severity == "CRITICAL":
        score += 18
        factors.append({"label": "Critical incident severity", "weight": 18})
    elif severity == "HIGH":
        score += 10
        factors.append({"label": "High incident severity", "weight": 10})
    if any(term in incident_type for term in ("ACCIDENT", "COLLISION")):
        score += 18
        factors.append({"label": "Accident or collision indicator", "weight": 18})
    if any(term in incident_type for term in ("PEDESTRIAN", "MOTORCYCLE", "VULNERABLE")):
        score += 8
        factors.append({"label": "Vulnerable road user indicator", "weight": 8})
    if len(objects) >= 5:
        score += 6
        factors.append({"label": "High detected object count", "weight": 6})
    if "traffic" in object_names or "density" in incident_type:
        score += 6
        factors.append({"label": "Traffic density indicator", "weight": 6})

    score = max(0, min(100, round(score)))
    calculated_severity = "CRITICAL" if score >= 75 else "HIGH" if score >= 50 else "MEDIUM" if score >= 25 else "LOW"
    confidence = round(max(0.5, min(0.99, float(incident.get("confidence") or 0.75))) * 100)
    return {
        "score": score,
        "severity": calculated_severity,
        "priority": SEVERITY_TO_PRIORITY[calculated_severity],
        "confidence": confidence,
        "factors": factors,
    }

def _response(row) -> dict[str, Any]:
    result = dict(row)
    result["factors"] = _json(result.get("factors"), [])
    result["incident"] = get_incident(result["incident_id"])
    result["assignments"] = list_assignments(result["id"])
    result["dispatch"] = dispatch_status()
    result["location"] = get_location_provider().get_location(result["incident_id"])

def create_or_get_response(incident_id: str) -> dict[str, Any] | None:
    incident = get_incident(incident_id)
    if not incident or incident.get("severity") not in {"HIGH", "CRITICAL"}:
        return None
    with get_connection() as conn:
        cursor = conn.cursor()
        cursor.execute("SELECT * FROM emergency_responses WHERE incident_id = ?", (incident_id,))
        existing = cursor.fetchone()
        if existing:
            return _response(existing)
        calc = _score_incident(incident)
        now = _now()
        response_id = _id("er")
        cursor.execute("""
            INSERT INTO emergency_responses
            (id, incident_id, severity, priority, score, confidence, factors, status, created_at, updated_at)
            VALUES (?, ?, ?, ?, ?, ?, ?, 'PENDING', ?, ?)
        """, (response_id, incident_id, calc["severity"], calc["priority"], calc["score"], calc["confidence"], json.dumps(calc["factors"]), now, now))
        cursor.execute("""
            INSERT INTO emergency_response_events
            (id, response_id, incident_id, event_type, status, actor, description, timestamp)
            VALUES (?, ?, ?, 'EMERGENCY_CREATED', 'PENDING', 'system', 'Emergency response case created from existing incident', ?)
        """, (_id("ere"), response_id, incident_id, now))
        conn.commit()
    return get_response(response_id)


def sync_responses() -> None:
    for incident in list_incidents(limit=500):
        if incident.get("severity") in {"HIGH", "CRITICAL"} and incident.get("status") != "RESOLVED":
            create_or_get_response(incident["id"])


def list_responses(status: str | None = None, severity: str | None = None, limit: int = 100) -> list[dict[str, Any]]:
    sync_responses()
    with get_connection() as conn:
        query = "SELECT * FROM emergency_responses WHERE 1=1"
        params: list[Any] = []
        if status and status != "ALL":
            query += " AND status = ?"
            params.append(status)
        if severity and severity != "ALL":
            query += " AND severity = ?"
            params.append(severity)
        query += " ORDER BY CASE priority WHEN 'P1' THEN 1 WHEN 'P2' THEN 2 WHEN 'P3' THEN 3 ELSE 4 END, score DESC, created_at DESC LIMIT ?"
        params.append(limit)
        cursor = conn.cursor()
        cursor.execute(query, params)
        return [_response(row) for row in cursor.fetchall()]

def transition_response(response_id: str, status: str, actor: str = "operator", description: str | None = None) -> dict[str, Any] | None:
    status = status.upper()
    if status not in STATUSES:
        return None
    with get_connection() as conn:
        cursor = conn.cursor()
        cursor.execute("SELECT * FROM emergency_responses WHERE id = ?", (response_id,))
        row = cursor.fetchone()
        if not row or status == row["status"] or status not in STATUS_TRANSITIONS.get(row["status"], set()):
            return None
        now = _now()
        column = {"ACKNOWLEDGED": "acknowledged_at", "DISPATCH_RECOMMENDED": "dispatch_recommended_at", "DISPATCHED": "dispatched_at", "RESPONDING": "responding_at", "ON_SCENE": "on_scene_at", "RESOLVED": "resolved_at", "CANCELLED": "cancelled_at"}.get(status)
        if column:
            cursor.execute(f"UPDATE emergency_responses SET status = ?, updated_at = ?, {column} = ? WHERE id = ?", (status, now, now, response_id))
        else:
            cursor.execute("UPDATE emergency_responses SET status = ?, updated_at = ? WHERE id = ?", (status, now, response_id))
        if status in {"RESOLVED", "CANCELLED"}:
            cursor.execute("""
                UPDATE emergency_units SET status = 'AVAILABLE', assigned_incident_id = NULL,
                eta_minutes = NULL, last_updated = ?
                WHERE assigned_incident_id = ?
            """, (now, row["incident_id"]))
            cursor.execute("UPDATE emergency_assignments SET released_at = ? WHERE response_id = ? AND released_at IS NULL", (now, response_id))
        cursor.execute("""
            INSERT INTO emergency_response_events
            (id, response_id, incident_id, event_type, status, actor, description, timestamp)
            VALUES (?, ?, ?, 'STATUS_CHANGED', ?, ?, ?, ?)
        """, (_id("ere"), row["incident_id"], row["incident_id"], status, actor, description or f"Response status changed to {status}", now))
        conn.commit()
    return get_response(response_id)


def list_units(status: str | None = None) -> list[dict[str, Any]]:
    with get_connection() as conn:
        cursor = conn.cursor()
        if status and status != "ALL":
            cursor.execute("SELECT * FROM emergency_units WHERE status = ? ORDER BY name", (status,))
        else:
            cursor.execute("SELECT * FROM emergency_units ORDER BY status, name")
        units = [dict(row) for row in cursor.fetchall()]
        for unit in units:
            source = unit.get("location_source") or ("SIMULATED" if unit.get("simulated") else "UNAVAILABLE")
            unit["location"] = {
                "label": unit.get("location") or "Location metadata unavailable",
                "latitude": unit.get("latitude"),
                "longitude": unit.get("longitude"),
                "accuracy": None,
                "source": source,
                "timestamp": unit.get("last_location_update"),
            }
            unit["location_source"] = source
            unit["dispatch_source"] = "SIMULATED" if unit.get("simulated") else "UNAVAILABLE"
        return units


def create_unit(payload: dict[str, Any]) -> dict[str, Any] | None:
    unit_type = str(payload.get("unit_type") or "").upper()
    status = str(payload.get("status") or "AVAILABLE").upper()
    if unit_type not in UNIT_TYPES or status not in UNIT_STATUSES or not payload.get("unit_id") or not payload.get("name"):
        return None
    now = _now()
    try:
        with get_connection() as conn:
            conn.execute("""
                INSERT INTO emergency_units
                (id, name, unit_type, status, location, eta_minutes, last_updated, simulated, latitude, longitude, location_source, last_location_update)
                VALUES (?, ?, ?, ?, ?, ?, ?, 1, ?, ?, ?, ?)
            """, (payload["unit_id"], payload["name"], unit_type, status, payload.get("location") or "Location metadata unavailable", payload.get("eta_minutes"), now, payload.get("latitude"), payload.get("longitude"), payload.get("location_source") or "UNAVAILABLE", now))
            conn.commit()
    except Exception:
        return None
    return next((unit for unit in list_units() if unit["id"] == payload["unit_id"]), None)


def update_unit(unit_id: str, payload: dict[str, Any]) -> dict[str, Any] | None:
    allowed = {"name", "unit_type", "status", "location", "eta_minutes", "latitude", "longitude", "location_source"}
    values = {key: value for key, value in payload.items() if key in allowed and value is not None}
    if "unit_type" in values:
        values["unit_type"] = str(values["unit_type"]).upper()
        if values["unit_type"] not in UNIT_TYPES:
            return None
    if "status" in values:
        values["status"] = str(values["status"]).upper()
        if values["status"] not in UNIT_STATUSES:
            return None
    if not values:
        return next((unit for unit in list_units() if unit["id"] == unit_id), None)
    values["last_updated"] = _now()
    assignments = ", ".join(f"{key} = ?" for key in values)
    try:
        with get_connection() as conn:
            cursor = conn.cursor()
            cursor.execute(f"UPDATE emergency_units SET {assignments} WHERE id = ?", (*values.values(), unit_id))
            if cursor.rowcount == 0:
                return None
            conn.commit()
    except Exception:
        return None
    return next((unit for unit in list_units() if unit["id"] == unit_id), None)


def list_assignments(response_id: str) -> list[dict[str, Any]]:

            SELECT a.*, u.name, u.unit_type, u.status AS unit_status
            FROM emergency_assignments a JOIN emergency_units u ON u.id = a.unit_id
            WHERE a.response_id = ? ORDER BY a.assigned_at DESC
        """, (response_id,))
        return [dict(row) for row in cursor.fetchall()]


def assign_unit(response_id: str, unit_id: str, actor: str = "operator") -> dict[str, Any] | None:
    response = get_response(response_id)
    if not response:
        return None
    try:
        dispatch_result = get_dispatch_provider().dispatch(response_id, unit_id, 10)
    except RuntimeError:
        return None
    with get_connection() as conn:
        cursor = conn.cursor()
        cursor.execute("SELECT * FROM emergency_units WHERE id = ?", (unit_id,))
        unit = cursor.fetchone()
        if not unit or unit["status"] != "AVAILABLE":
            return None
        cursor.execute("SELECT id FROM emergency_assignments WHERE response_id = ? AND released_at IS NULL", (response_id,))
        if cursor.fetchone():
            return None
        now = _now()
        cursor.execute("UPDATE emergency_units SET status = 'DISPATCHED', assigned_incident_id = ?, eta_minutes = ?, last_updated = ? WHERE id = ?", (response["incident_id"], 10, now, unit_id))
        cursor.execute("INSERT INTO emergency_assignments (id, response_id, incident_id, unit_id, assigned_by, assigned_at) VALUES (?, ?, ?, ?, ?, ?)", (_id("ass"), response_id, response["incident_id"], unit_id, actor, now))
        cursor.execute("INSERT INTO emergency_response_events (id, response_id, incident_id, event_type, status, actor, description, timestamp) VALUES (?, ?, ?, 'UNIT_ASSIGNED', ?, ?, ?, ?)", (_id("ere"), response_id, response["incident_id"], response["status"], actor, f"{dispatch_result['mode']} dispatch requested for unit {unit_id}", now))
        conn.commit()
    if response["status"] == "DISPATCH_RECOMMENDED":
        transition_response(response_id, "DISPATCHED", actor, "Unit assignment dispatched")
    return get_response(response_id)

def get_recommendations(response_id: str) -> list[dict[str, Any]]:
    response = get_response(response_id)
    if not response:
        return []
    incident = response.get("incident") or {}
    actions = [
        ("Preserve evidence", "Maintain the existing incident evidence chain", "P1" if response["priority"] == "P1" else "P2"),
        ("Monitor adjacent cameras", "Increase situational awareness around the incident camera", "P2"),
        ("Open investigation workflow", "Review the incident timeline and evidence", "P2"),
    ]
    if response["severity"] in {"HIGH", "CRITICAL"}:
        actions.insert(0, ("Dispatch recommended unit", "Severity and risk score warrant operator dispatch review", response["priority"]))
    if "PEDESTRIAN" in str(incident.get("incident_type", "")):
        actions.insert(0, ("Dispatch medical response", "Vulnerable road user indicator is present", "P1"))
    return [{"action": a, "reason": r, "priority": p, "status": "RECOMMENDED", "timestamp": response["created_at"]} for a, r, p in actions]


def analytics() -> dict[str, Any]:
    sync_responses()
    responses = list_responses(limit=1000)
    if not responses:
        return {"total": 0, "message": "Insufficient historical data", "severity_distribution": {}, "status_distribution": {}}
    durations = []
    ack_times = []
    for item in responses:
        created = datetime.fromisoformat(item["created_at"])
        if item.get("acknowledged_at"):
            ack_times.append(max(0, (datetime.fromisoformat(item["acknowledged_at"]) - created).total_seconds()))
        if item.get("resolved_at"):
            durations.append(max(0, (datetime.fromisoformat(item["resolved_at"]) - created).total_seconds()))
    return {
        "total": len(responses),
        "critical": sum(1 for r in responses if r["severity"] == "CRITICAL"),
        "average_acknowledgement_seconds": round(sum(ack_times) / len(ack_times)) if ack_times else None,
        "average_resolution_seconds": round(sum(durations) / len(durations)) if durations else None,
        "severity_distribution": dict(Counter(r["severity"] for r in responses)),
        "status_distribution": dict(Counter(r["status"] for r in responses)),
        "message": None if ack_times or durations else "Insufficient historical data",
    }
